    Interface Problem

    The RFC 9457 body every error answer has (Problem in the contract). type and title follow from code; instance and traceId are filled from the request, so a log line and an answer can be matched.

    interface Problem {
        type: string;
        title: string;
        status: number;
        detail?: string;
        instance?: string;
        code:
            | "MALFORMED_REQUEST"
            | "VALIDATION_ERROR"
            | "UNAUTHENTICATED"
            | "FORBIDDEN"
            | "NOT_FOUND"
            | "METHOD_NOT_ALLOWED"
            | "SLOT_UNAVAILABLE"
            | "DUPLICATE_RESOURCE"
            | "INVALID_STATE_TRANSITION"
            | "PAYMENT_ALREADY_RECORDED"
            | "APPOINTMENT_NOT_COMPLETED"
            | "MATERIAL_STOCK_INSUFFICIENT"
            | "INVALID_PERIOD"
            | "BUSINESS_RULE_VIOLATION"
            | "UNSUPPORTED_MEDIA_TYPE"
            | "INTERNAL_ERROR";
        traceId?: string;
        errors?: ValidationErrorItem[];
    }
